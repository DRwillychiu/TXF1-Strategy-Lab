{STRATEGY_GEN_BearBounceSell}
{Batch02 - S8 | 空頭反彈放空}
{操作週期: 15分鐘 + 日線(Data2) | 方向: ★純做空}
{賺什麼錢: 空頭趨勢中的急彈反彈放空再跌}
{指數位階論: 高位階空頭反彈幅度大，ATR自適應停損}

Inputs:
    RSILen(6),              {RSI計算週期}
    RSIOverbought(85),      {RSI超買門檻}
    ConsecUpBars(3),        {連續上漲K棒數}
    DevATRMult(2.5),        {偏離ATR倍數門檻}
    ATRLen(14),             {ATR計算週期}
    MA_Len(20),             {Data2趨勢均線}
    StopATRMult(2.0),       {停損ATR倍數}
    TargetATRMult(3.0),     {停利ATR倍數}
    SessionStart(0845),
    SessionEnd(1330);

Variables:
    v_RSI(0),
    v_ATR(0),
    v_Deviation(0),
    v_ConsecUp(0),
    v_TrendFilter(False),
    v_StopPrice(0),
    v_TargetPrice(0),
    v_Prev_MP(0);

v_Prev_MP = MarketPosition[1];

v_RSI = RSI(Close, RSILen);
v_ATR = AvgTrueRange(ATRLen);

{連續上漲計算}
v_ConsecUp = 0;
If Close > Close[1] Then v_ConsecUp = v_ConsecUp + 1;
If Close[1] > Close[2] Then v_ConsecUp = v_ConsecUp + 1;
If Close[2] > Close[3] Then v_ConsecUp = v_ConsecUp + 1;

{偏離度: 收盤偏離MA的ATR倍數(向上偏離)}
If v_ATR > 0 Then
    v_Deviation = (Close - Average(Close, MA_Len)) / v_ATR;

{Data2(日線)趨勢過濾: 收盤在MA20之下=空頭}
v_TrendFilter = Close of Data2 < Average(Close of Data2, MA_Len)[1] of Data2;

{進場: RSI極度超買 + 連續上漲 + 偏離過大 + 日線空頭}
If v_Prev_MP = 0 And
   Time >= SessionStart And Time <= SessionEnd And
   v_RSI > RSIOverbought And
   v_ConsecUp >= ConsecUpBars And
   v_Deviation >= DevATRMult And
   v_TrendFilter Then Begin
    SellShort ("BBS_Short") Next Bar at Market;
End;

{出場管理}
If MarketPosition = -1 Then Begin
    v_StopPrice = EntryPrice + v_ATR * StopATRMult;
    v_TargetPrice = EntryPrice - v_ATR * TargetATRMult;
    
    BuyToCover ("BBS_Stop") Next Bar at v_StopPrice Stop;
    BuyToCover ("BBS_Target") Next Bar at v_TargetPrice Limit;
    
    If Time >= 1340 Then
        BuyToCover ("BBS_EOD") Next Bar at Market;
End;
