{STRATEGY_GEN_AdaptiveBreakout}
{Category: Index-Level Adaptive Channel Breakout}
{Direction: Bidirectional (Long + Short)}
{Timeframe: 30-min on TXF1}
{N-bar HH/LL breakout with MinList(ATR*mult, Close*pct%) adaptive stop}

inputs:
    BreakoutBars(20),
    ATRLen(14),
    StopATRMult(2.0),
    StopPctCap(0.015),
    TargetATRMult(3.0),
    MinVolRatio(1.0),
    SessionStart(0845),
    SessionEnd(1330);

variables:
    v_ATR(0),
    v_HH(0),
    v_LL(0),
    v_StopATR(0),
    v_StopPct(0),
    v_AdaptiveStop(0),
    v_StopPrice(0),
    v_TargetPrice(0),
    v_VolOK(False),
    v_Prev_MP(0);

{ ---- Previous bar position tracking ---- }
v_Prev_MP = MarketPosition[1];

{ ---- Indicator calculations ---- }
v_ATR = AvgTrueRange(ATRLen);
v_HH = Highest(High, BreakoutBars);
v_LL = Lowest(Low, BreakoutBars);

{ Adaptive stop: smaller of ATR-based and percentage-based }
v_StopATR = v_ATR * StopATRMult;
v_StopPct = Close * StopPctCap;
v_AdaptiveStop = MinList(v_StopATR, v_StopPct);

{ Volume filter }
v_VolOK = True;
If Average(Volume, 20) > 0 Then
    v_VolOK = Volume >= Average(Volume, 20) * MinVolRatio;

{ ---- Long Entry: breakout above N-bar high ---- }
If v_Prev_MP = 0 And
   Time >= SessionStart And Time <= SessionEnd And
   Close > v_HH[1] And
   v_VolOK Then Begin
    Buy ("LE_AB_Long") Next Bar at Market;
End;

{ ---- Short Entry: breakout below N-bar low ---- }
If v_Prev_MP = 0 And
   Time >= SessionStart And Time <= SessionEnd And
   Close < v_LL[1] And
   v_VolOK Then Begin
    SellShort ("SE_AB_Short") Next Bar at Market;
End;

{ ---- Long Exit management ---- }
If MarketPosition = 1 Then Begin
    v_StopPrice = EntryPrice - v_AdaptiveStop;
    v_TargetPrice = EntryPrice + v_ATR * TargetATRMult;

    Sell ("LX_AB_SL") Next Bar at v_StopPrice Stop;
    Sell ("LX_AB_TP") Next Bar at v_TargetPrice Limit;

    If Time >= 1340 Then
        Sell ("LX_AB_EOD") Next Bar at Market;
End;

{ ---- Short Exit management ---- }
If MarketPosition = -1 Then Begin
    v_StopPrice = EntryPrice + v_AdaptiveStop;
    v_TargetPrice = EntryPrice - v_ATR * TargetATRMult;

    BuyToCover ("SX_AB_SL") Next Bar at v_StopPrice Stop;
    BuyToCover ("SX_AB_TP") Next Bar at v_TargetPrice Limit;

    If Time >= 1340 Then
        BuyToCover ("SX_AB_EOD") Next Bar at Market;
End;

{ ---- Track position state ---- }
v_Prev_MP = MarketPosition;
